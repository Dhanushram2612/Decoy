FROM python:3.10-slim

WORKDIR /app

# system deps needed for some ML wheels + tldextract's suffix list fetch
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    curl \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# pre-fetch tldextract's public suffix list at build time, not first request
RUN python -c "import tldextract; tldextract.extract('https://example.com')"

COPY . .

# HF Spaces expects the app on port 7860
EXPOSE 7860

CMD ["streamlit", "run", "app.py", "--server.port=7860", "--server.address=0.0.0.0"]
